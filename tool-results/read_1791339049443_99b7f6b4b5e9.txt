   900→   900→  function tokenizeExpr(expr) {
   901→   901→    var tokens = [];
   902→   902→    var i = 0, len = expr.length;
   903→   903→    while (i < len) {
   904→   904→      var c = expr[i];
   905→   905→      if (c === ' ' || c === '\t' || c === '\n' || c === '\r') { i++; continue; }
   906→   906→      // Number literal — digits and at most one decimal point.
   907→   907→      if ((c >= '0' && c <= '9') || c === '.') {
   908→   908→        var num = '';
   909→   909→        var dotSeen = false;
   910→   910→        while (i < len && ((expr[i] >= '0' && expr[i] <= '9') || expr[i] === '.')) {
   911→   911→          if (expr[i] === '.') {
   912→   912→            if (dotSeen) break;
   913→   913→            dotSeen = true;
   914→   914→          }
   915→   915→          num += expr[i]; i++;
   916→   916→        }
   917→   917→        var n = parseFloat(num);
   918→   918→        if (isNaN(n)) throw new Error('Invalid number in expression: ' + num);
   919→   919→        tokens.push({ type: 'num', value: n });
   920→   920→        continue;
   921→   921→      }
   922→   922→      // Single-quoted string literal — doubled '' inside means an escaped quote.
   923→   923→      if (c === "'") {
   924→   924→        var str = '';
   925→   925→        i++; // skip opening quote
   926→   926→        while (i < len && expr[i] !== "'") { str += expr[i]; i++; }
   927→   927→        if (i >= len) throw new Error('Unterminated string literal in expression');
   928→   928→        i++; // skip closing quote
   929→   929→        tokens.push({ type: 'str', value: str });
   930→   930→        continue;
   931→   931→      }
   932→   932→      // Identifier — letters, digits, underscore (must start with letter/underscore).
   933→   933→      if ((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || c === '_') {
   934→   934→        var id = '';
   935→   935→        while (i < len && ((expr[i] >= 'a' && expr[i] <= 'z') ||
   936→   936→                           (expr[i] >= 'A' && expr[i] <= 'Z') ||
   937→   937→                           (expr[i] >= '0' && expr[i] <= '9') || expr[i] === '_')) {
   938→   938→          id += expr[i]; i++;
   939→   939→        }
   940→   940→        tokens.push({ type: 'id', value: id });
   941→   941→        continue;
   942→   942→      }
   943→   943→      // Operators + parens.
   944→   944→      if ('+-*/%()'.indexOf(c) >= 0) { tokens.push({ type: 'op', value: c }); i++; continue; }
   945→   945→      throw new Error('Invalid character in expression: ' + c);
   946→   946→    }
   947→   947→    return tokens;
   948→   948→  }
   949→   949→
   950→   950→  // Recursive-descent parser → AST nodes:
   951→   951→  //   { type: 'literal', value: <number|string> }
   952→   952→  //   { type: 'column',  name: <string> }
   953→   953→  //   { type: 'binop',   op: '+'|'-'|'*'|'/'|'%', left: <node>, right: <node> }
   954→   954→  function parseExpr(tokens) {
   955→   955→    var pos = 0;
   956→   956→    function peek() { return tokens[pos]; }
   957→   957→    function next() { return tokens[pos++]; }
   958→   958→    function parseExpression() {
   959→   959→      var node = parseTerm();
   960→   960→      while (peek() && peek().type === 'op' && (peek().value === '+' || peek().value === '-')) {
   961→   961→        var op = next().value;
   962→   962→        var right = parseTerm();
   963→   963→        node = { type: 'binop', op: op, left: node, right: right };
   964→   964→      }
   965→   965→      return node;
   966→   966→    }
   967→   967→    function parseTerm() {
   968→   968→      var node = parseFactor();
   969→   969→      while (peek() && peek().type === 'op' &&
   970→   970→             (peek().value === '*' || peek().value === '/' || peek().value === '%')) {
   971→   971→        var op = next().value;
   972→   972→        var right = parseFactor();
   973→   973→        node = { type: 'binop', op: op, left: node, right: right };
   974→   974→      }
   975→   975→      return node;
   976→   976→    }
   977→   977→    function parseFactor() {
   978→   978→      var t = peek();
   979→   979→      if (!t) throw new Error('Unexpected end of expression');
   980→   980→      if (t.type === 'num' || t.type === 'str') { next(); return { type: 'literal', value: t.value }; }
   981→   981→      if (t.type === 'id') { next(); return { type: 'column', name: t.value }; }
   982→   982→      if (t.type === 'op' && t.value === '(') {
   983→   983→        next();
   984→   984→        var node = parseExpression();
   985→   985→        if (!peek() || peek().value !== ')') throw new Error('Missing closing parenthesis');
   986→   986→        next(); // consume ')'
   987→   987→        return node;
   988→   988→      }
   989→   989→      throw new Error('Unexpected token in expression: ' + t.value);
   990→   990→    }
   991→   991→    var ast = parseExpression();
   992→   992→    if (pos < tokens.length) throw new Error('Unexpected trailing tokens in expression');
   993→   993→    return ast;
   994→   994→  }
   995→   995→
   996→   996→  // Evaluate an AST node against a row object. The '+' operator is polymorphic:
   997→   997→  // if either operand is a string, both sides stringify and concatenate; else
   998→   998→  // numeric addition. All other operators are numeric only.
   999→   999→  function evalAst(node, row) {
  1000→  1000→    if (node.type === 'literal') return node.value;
  1001→  1001→    if (node.type === 'column') {
  1002→  1002→      if (!(node.name in row)) throw new Error('Unknown column in expression: ' + node.name);
  1003→  1003→      return row[node.name];
  1004→  1004→    }
  1005→  1005→    if (node.type === 'binop') {
  1006→  1006→      var l = evalAst(node.left, row);
  1007→  1007→      var r = evalAst(node.right, row);
  1008→  1008→      if (node.op === '+' && (typeof l === 'string' || typeof r === 'string')) {
  1009→  1009→        return String(l) + String(r);
  1010→  1010→      }
  1011→  1011→      var ln = Number(l), rn = Number(r);
  1012→  1012→      if (isNaN(ln) || isNaN(rn)) throw new Error('Non-numeric operand in arithmetic expression');
  1013→  1013→      switch (node.op) {
  1014→  1014→        case '+': return ln + rn;
  1015→  1015→        case '-': return ln - rn;
  1016→  1016→        case '*': return ln * rn;
  1017→  1017→        case '/': return rn === 0 ? 0 : ln / rn;
  1018→  1018→        case '%': return rn === 0 ? 0 : ln % rn;
  1019→  1019→      }
  1020→  1020→      throw new Error('Unknown operator: ' + node.op);
  1021→  1021→    }
  1022→  1022→    throw new Error('Cannot evaluate AST node: ' + node.type);
  1023→  1023→  }
  1024→  1024→
  1025→  1025→  // Public façade — tokenize + parse + evaluate. AST is cached per expression
  1026→  1026→  // (via the closure on the parsed result) so per-row evaluation skips parsing.
  1027→  1027→  function makeExprEvaluator(expression) {
  1028→  1028→    var tokens = tokenizeExpr(expression);
  1029→  1029→    var ast = parseExpr(tokens);
  1030→  1030→    return function (row) { return evalAst(ast, row); };
  1031→  1031→  }

  1033→  1033→  // ─── Case-insensitive column resolver ──────────────────────────────────────
  1034→  1034→  // Groq frequently returns column names in a DIFFERENT CASE than the actual
  1035→  1035→  // data — e.g. valueColumn: "QTY" when the data has lowercase "qty", or
  1036→  1036→  // "Revenue" when the data has "revenue". Without a resolver, r["QTY"]
  1037→  1037→  // returns undefined → Number(undefined) is NaN → NaN propagates through
  1038→  1038→  // sum/avg/min/max → the entire result column shows as NaN.
  1039→  1039→  //
  1040→  1040→  // The resolver: exact match first, case-insensitive match second, falls
  1041→  1041→  // back to the original name (so undefined values surface the bug visibly
  1042→  1042→  // in the result — better than silently failing).
  1043→  1043→  var _colLookupCache = {};
  1044→  1044→  function resolveColumn(name, docColumns) {
  1045→  1045→    if (!name) return name;
  1046→  1046→    if (docColumns.indexOf(name) >= 0) return name;
  1047→  1047→    var cacheKey = name.toLowerCase() + '\u0001' + docColumns.join('\u0001');
  1048→  1048→    if (_colLookupCache[cacheKey]) return _colLookupCache[cacheKey];
  1049→  1049→    var lower = name.toLowerCase();
  1050→  1050→    for (var i = 0; i < docColumns.length; i++) {
  1051→  1051→      if (docColumns[i].toLowerCase() === lower) {
  1052→  1052→        _colLookupCache[cacheKey] = docColumns[i];
  1053→  1053→        return docColumns[i];
  1054→  1054→      }
  1055→  1055→    }
  1056→  1056→    _colLookupCache[cacheKey] = name;
  1057→  1057→    return name;
  1058→  1058→  }

  1130→  1130→  // ─── Pipeline executor ─────────────────────────────────────────────────────
  1131→  1131→  // Runs each step in order over a shallow-cloned row array. Each step mutates
  1132→  1132→  // the array (filter shortens, summarize reshapes, sort reorders, etc.). The
  1133→  1133→  // final shape is returned to the renderer.
  1134→  1134→  function executePipeline(doc, spec) {
  1135→  1135→    var rows = (doc.rows || []).slice();
  1136→  1136→    var docCols = doc.columns || (rows.length ? Object.keys(rows[0]) : []);
  1137→  1137→    _colLookupCache = {};  // reset per-execution
  1138→  1138→    (spec.steps || []).forEach(function (step) {
  1139→  1139→      if (step.type === 'filter') {
  1140→  1140→        var fCol = resolveColumn(step.column, docCols);
  1141→  1141→        rows = rows.filter(function (r) {
  1142→  1142→          var v = r[fCol];
  1143→  1143→          switch (step.operator) {
  1144→  1144→            case '==':  return String(v) == String(step.value);
  1145→  1145→            case '!=':  return String(v) != String(step.value);
  1146→  1146→            case '>':   return Number(v) >  Number(step.value);
  1147→  1147→            case '<':   return Number(v) <  Number(step.value);
  1148→  1148→            case '>=':  return Number(v) >= Number(step.value);
  1149→  1149→            case '<=':  return Number(v) <= Number(step.value);
  1150→  1150→            case 'contains':
  1151→  1151→              return String(v).toLowerCase().indexOf(String(step.value).toLowerCase()) >= 0;
  1152→  1152→            default: return true;
  1153→  1153→          }
  1154→  1154→        });
  1155→  1155→      } else if (step.type === 'summarize') {
  1156→  1156→        var groupByRaw = step.groupBy && step.groupBy.length ? step.groupBy : [];
  1157→  1157→        var groupBy = groupByRaw.map(function (c) { return resolveColumn(c, docCols); });
  1158→  1158→        var valCol = step.valueColumn ? resolveColumn(step.valueColumn, docCols) : null;
  1159→  1159→        var groups = {};
  1160→  1160→        var order = [];
  1161→  1161→        rows.forEach(function (r) {
  1162→  1162→          var key = groupBy.map(function (c) { return r[c]; }).join(' | ');
  1163→  1163→          if (!groups[key]) { groups[key] = { count: 0, sum: 0, min: Infinity, max: -Infinity }; order.push(key); }
  1164→  1164→          var g = groups[key];
  1165→  1165→          var val = valCol ? Number(r[valCol]) : 0;
  1166→  1166→          if (isNaN(val)) val = 0;  // guard against NaN propagation
  1167→  1167→          g.count++;
  1168→  1168→          if (valCol) {
  1169→  1169→            g.sum += val;
  1170→  1170→            g.min = Math.min(g.min, val);
  1171→  1171→            g.max = Math.max(g.max, val);
  1172→  1172→          }
  1173→  1173→        });
  1174→  1174→        rows = order.map(function (key) {
  1175→  1175→          var g = groups[key];
  1176→  1176→          var parts = key.split(' | ');
  1177→  1177→          var row = {};
  1178→  1178→          groupBy.forEach(function (c, i) { row[c] = parts[i]; });
  1179→  1179→          if (valCol) {
  1180→  1180→            var agg = step.aggregation || 'sum';
  1181→  1181→            row[agg + '_of_' + valCol] =
  1182→  1182→              agg === 'sum'   ? g.sum :
  1183→  1183→              agg === 'count' ? g.count :
  1184→  1184→              agg === 'avg'   ? (g.count ? g.sum / g.count : 0) :
  1185→  1185→              agg === 'min'   ? (g.min === Infinity ? 0 : g.min) :
  1186→  1186→              agg === 'max'   ? (g.max === -Infinity ? 0 : g.max) : null;
  1187→  1187→          } else {
  1188→  1188→            row.count = g.count;
  1189→  1189→          }
  1190→  1190→          return row;
  1191→  1191→        });
  1192→  1192→      } else if (step.type === 'sort') {
  1193→  1193→        var sCol = resolveColumn(step.column, docCols);
  1194→  1194→        var order2 = step.order === 'desc' ? -1 : 1;
  1195→  1195→        rows.sort(function (a, b) {
  1196→  1196→          var av = a[sCol], bv = b[sCol];
  1197→  1197→          if (typeof av === 'number' && typeof bv === 'number') return (av - bv) * order2;
  1198→  1198→          return order2 * String(av).localeCompare(String(bv));
  1199→  1199→        });
  1200→  1200→      } else if (step.type === 'limit') {
  1201→  1201→        rows = rows.slice(0, Math.max(0, step.count | 0));
  1202→  1202→      } else if (step.type === 'select') {
  1203→  1203→        var selColsRaw = step.columns || [];
  1204→  1204→        var selCols = selColsRaw.map(function (c) { return resolveColumn(c, docCols); });
  1205→  1205→        rows = rows.map(function (r) {
  1206→  1206→          var o = {};
  1207→  1207→          selCols.forEach(function (c) { o[c] = r[c]; });
  1208→  1208→          return o;
  1209→  1209→        });
  1210→  1210→      } else if (step.type === 'transform') {
  1211→  1211→        // Build the evaluator ONCE, then map — parsing per row would be wasteful.
  1212→  1212→        var evalFn;
  1213→  1213→        try { evalFn = makeExprEvaluator(step.expression); }
  1214→  1214→        catch (e) { throw new Error('Transform step has an invalid expression: ' + e.message); }
  1215→  1215→        var newCol = step.newColumn || 'computed';
  1216→  1216→        rows = rows.map(function (r) {
  1217→  1217→          var o = Object.assign({}, r);
  1218→  1218→          try { o[newCol] = evalFn(r); } catch (e) { o[newCol] = ''; }
  1219→  1219→          return o;
  1220→  1220→        });
  1221→  1221→      }
  1222→  1222→      // Unknown step types are silently skipped — the spec validator already
  1223→  1223→      // filtered out garbage, so this is a defensive no-op.
  1224→  1224→    });
  1225→  1225→    return { rows: rows, columns: rows.length ? Object.keys(rows[0]) : [] };
  1226→  1226→  }

   440→   440→  // CSV — native parse, no library. Handles quoted fields with embedded commas,
   441→   441→  // newlines, and doubled-quote escapes (RFC 4180).
   442→   442→  function parseCSV(file) {
   443→   443→    return new Promise(function (resolve, reject) {
   444→   444→      var reader = new FileReader();

   461→   461→      };

   465→   465→  // Minimal RFC-4180 CSV parser → { rows: [{...}], columns: [...] }
   466→   466→  function csvToRows(text) {
   467→   467→    var records = [];
   468→   468→    var field = '';
   469→   469→    var row = [];
   470→   470→    var i = 0;
   471→   471→    var inQuotes = false;
   472→   472→    while (i < text.length) {
   473→   473→      var c = text[i];
   474→   474→      if (inQuotes) {
   475→   475→        if (c === '"') {
   476→   476→          if (text[i + 1] === '"') { field += '"'; i += 2; continue; }
   477→   477→          inQuotes = false; i++; continue;
   478→   478→        }
   479→   479→        field += c; i++; continue;
   480→   480→      }
   481→   481→      if (c === '"') { inQuotes = true; i++; continue; }
   482→   482→      if (c === ',') { row.push(field); field = ''; i++; continue; }
   483→   483→      if (c === '\r') { i++; continue; }
   484→   484→      if (c === '\n') { row.push(field); records.push(row); row = []; field = ''; i++; continue; }
   485→   485→      field += c; i++;
   486→   486→    }
   487→   487→    if (field.length || row.length) { row.push(field); records.push(row); }
   488→   488→    if (!records.length) return { rows: [], columns: [] };
   489→   489→    var headers = records[0].map(function (h, idx) { return h || ('col_' + (idx + 1)); });
   490→   490→    var objRows = records.slice(1).filter(function (r) {
   491→   491→      return r.some(function (v) { return v !== '' && v != null; });
   492→   492→    }).map(function (r) {
   493→   493→      var o = {};
   494→   494→      headers.forEach(function (h, idx) {
   495→   495→        var v = r[idx];
   496→   496→        // Auto-coerce numeric strings — keeps summarize/sort working.
   497→   497→        if (v != null && v !== '' && !isNaN(v) && /^-?\d+(\.\d+)?$/.test(v.trim())) o[h] = Number(v);
   498→   498→        else o[h] = v != null ? v : '';
   499→   499→      });
   500→   500→      return o;
   501→   501→    });
   502→   502→    return { rows: objRows, columns: headers };
   503→   503→  }

   545→   545→  // Word (.docx) via mammoth. Extract raw text + try to parse the first HTML
   546→   546→  // <table> for tabular data. If no table found, treat as text-only doc.
   547→   547→  function parseWord(file) {
   548→   548→    return loadScript('mammoth').then(function () {
   549→   549→      return new Promise(function (resolve, reject) {
   550→   550→        var reader = new FileReader();
   551→   551→        reader.onerror = function () { reject(new Error('Couldn\u2019t read this Word file.')); };
   552→   552→        reader.onload = function () {

   559→   559→              mammoth.convertToHtml({ arrayBuffer: ab })
   560→   560→          ]).then(function (results) {
   561→   561→            var rawText = (results[0] && results[0].value) || '';

   570→   570→              }));

   572→   572→              warnings.push('No tables detected — this document is text-only. Pipeline steps that need rows (filter/summarize/sort) will return empty results, but the AI can still analyze the text.');
   573→   573→              resolve(makeDoc({
   574→   574→                type: 'word', name: file.name, fileSize: file.size,
   575→   575→                rows: [], columns: [], rawText: rawText, warnings: warnings
   576→   576→              }));
   577→   577→            }
   578→   578→          }).catch(function () {
   579→   579→            reject(new Error('Couldn\u2019t parse this Word file. Make sure it\u2019s a valid .docx (legacy .doc is not supported) and try again.'));
   580→   580→          });

   583→   583→      });

   586→   586→  // Parse the first <table> in an HTML string into { rows, columns }.
   587→   587→  function parseFirstHtmlTable(html) {
   588→   588→    if (!html) return { rows: [], columns: [] };
   589→   589→    var tmp = document.createElement('div');
   590→   590→    tmp.innerHTML = html;
   591→   591→    var table = tmp.querySelector('table');
   592→   592→    if (!table) return { rows: [], columns: [] };
   593→   593→    var trs = table.querySelectorAll('tr');
   594→   594→    if (!trs.length) return { rows: [], columns: [] };
   595→   595→    var headers = [];
   596→   596→    var firstCells = trs[0].querySelectorAll('th,td');
   597→   597→    firstCells.forEach(function (c, idx) { headers.push((c.textContent || '').trim() || 'col_' + (idx + 1)); });
   598→   598→    // If first row has <th>, treat it as header; otherwise the first row is data.
   599→   599→    var startIdx = trs[0].querySelector('th') ? 1 : 0;
   600→   600→    var rows = [];
   601→   601→    for (var i = startIdx; i < trs.length; i++) {
   602→   602→      var cells = trs[i].querySelectorAll('td,th');
   603→   603→      if (!cells.length) continue;
   604→   604→      var o = {};
   605→   605→      headers.forEach(function (h, idx) {
   606→   606→        var v = (cells[idx] ? cells[idx].textContent : '').trim();
   607→   607→        if (v !== '' && !isNaN(v) && /^-?\d+(\.\d+)?$/.test(v)) o[h] = Number(v);
   608→   608→        else o[h] = v;
   609→   609→      });
   610→   610→      rows.push(o);
   611→   611→    }
   612→   612→    return { rows: rows, columns: headers };
   613→   613→  }

   615→   615→  // PDF (.pdf) via pdf.js. Extracts text per page. Attempts a simple table
   616→   616→  // detection: if multiple text items share the same Y coordinate across pages,
   617→   617→  // group them into rows. If detection fails, fall back to text-only.
   618→   618→  function parsePDF(file) {
   619→   619→    return loadScript('pdfjs').then(function () {
   620→   620→      return new Promise(function (resolve, reject) {
   621→   621→        var reader = new FileReader();
   622→   622→        reader.onerror = function () { reject(new Error('Couldn\u2019t read this PDF.')); };
   623→   623→        reader.onload = function () {
   624→   624→          try {
   625→   625→            // Configure the worker — pdf.js won't run without it.
   626→   626→            if (window.pdfjsLib) {
   627→   627→              pdfjsLib.GlobalWorkerOptions.workerSrc = LIB_URLS.pdfjsWorker;
   628→   628→            }
   629→   629→            var loadingTask = pdfjsLib.getDocument({ data: reader.result });
   630→   630→            loadingTask.promise.then(function (pdf) {
   631→   631→              var numPages = pdf.numPages;
   632→   632→              var allItems = [];   // { page, str, x, y, w }
   633→   633→              var rawText  = [];
   634→   634→              // Sequentially fetch each page's text content. We do this serially
   635→   635→              // (rather than Promise.all) because pdf.js's getTextContent can
   636→   636→              // interfere with concurrent page operations on some builds, and
   637→   637→              // serial is plenty fast for the demo's <5MB PDFs.
   638→   638→              function readPagesSequentially(pageNum) {
   639→   639→                if (pageNum > numPages) return Promise.resolve();
   640→   640→                return pdf.getPage(pageNum).then(function (page) {
   641→   641→                  return page.getTextContent().then(function (tc) {
   642→   642→                    (tc.items || []).forEach(function (it) {
   643→   643→                      allItems.push({
   644→   644→                        page: pageNum,
   645→   645→                        str: it.str,
   646→   646→                        x: it.transform ? it.transform[4] : 0,
   647→   647→                        y: it.transform ? it.transform[5] : 0,
   648→   648→                        w: it.width || 0
   649→   649→                      });
   650→   650→                      if (it.str) rawText.push(it.str);
   651→   651→                    });
   652→   652→                    return readPagesSequentially(pageNum + 1);
   653→   653→                  });
   654→   654→                });
   655→   655→              }
   656→   656→              return readPagesSequentially(1).then(function () {
   657→   657→                var text = rawText.join(' ').replace(/\s+/g, ' ').trim();
   658→   658→                var parsed = detectPDFTable(allItems);
   659→   659→                var warnings = [];
   660→   660→                if (parsed.rows.length) {
   661→   661→                  warnings.push('Detected a tabular layout from text positioning — rows may be imperfect. Verify against the source PDF.');
   662→   662→                } else if (text) {
   663→   663→                  warnings.push('No clear table layout detected — treating as a text document. Pipeline steps that need rows will return empty results, but the AI can still analyze the text.');
   664→   664→                }
   665→   665→                resolve(makeDoc({
   666→   666→                  type: 'pdf', name: file.name, fileSize: file.size,
   667→   667→                  rows: parsed.rows, columns: parsed.columns,
   668→   668→                  rawText: text || '(no extractable text)', warnings: warnings
   669→   669→                }));
   670→   670→              });
   671→   671→            }).catch(function () {
   672→   672→              reject(new Error('Couldn\u2019t open this PDF. The file may be corrupted, password-protected, or a scanned image (text-extractable PDFs only).'));
   673→   673→            });
   674→   674→          } catch (e) {
   675→   675→            reject(new Error('Couldn\u2019t parse this PDF. Make sure it\u2019s a valid .pdf and try again.'));
   676→   676→          }
   677→   677→        };
   678→   678→        reader.readAsArrayBuffer(file);
   679→   679→      });
   680→   680→    });
   681→   681→  }

   683→   683→  // Heuristic PDF table detector: group items by Y coordinate (rounded to
   684→   684→  // nearest 2px to absorb sub-pixel jitter), sort each row's items by X, and
   685→   685→  // join into cells. If at least 3 rows × 2 cols, treat as a table.
   686→   686→  function detectPDFTable(items) {
   687→   687→    if (!items || items.length < 6) return { rows: [], columns: [] };
   688→   688→    // Group by rounded Y.
   689→   689→    var byY = {};
   690→   690→    items.forEach(function (it) {
   691→   691→      if (!it.str) return;
   692→   692→      var y = Math.round(it.y / 2) * 2;
   693→   693→      if (!byY[y]) byY[y] = [];
   694→   694→      byY[y].push(it);
   695→   695→    });
   696→   696→    var yKeys = Object.keys(byY).map(Number).sort(function (a, b) { return b - a; }); // top of page = larger Y
   697→   697→    if (yKeys.length < 3) return { rows: [], columns: [] };
   698→   698→    // Build row arrays (items sorted by X within each row).
   699→   699→    var rowArrays = yKeys.map(function (y) {
   700→   700→      return byY[y].sort(function (a, b) { return a.x - b.x; })
   701→   701→        .map(function (it) { return it.str; })
   702→   702→        .join(' ').trim();
   703→   703→    }).filter(function (s) { return s.length; });
   704→   704→    if (rowArrays.length < 3) return { rows: [], columns: [] };
   705→   705→    // Split each row on 2+ space gaps into columns. If the first row splits into
   706→   706→    // ≥2 columns, use that count as the column count.
   707→   707→    var splitRows = rowArrays.map(function (r) {
   708→   708→      return r.split(/\s{2,}/).map(function (s) { return s.trim(); }).filter(function (s) { return s.length; });
   709→   709→    });
   710→   710→    var colCount = splitRows[0].length;
   711→   711→    if (colCount < 2) return { rows: [], columns: [] };
   712→   712→    // Require that most rows have ≥2 columns; otherwise it's just paragraph text.
   713→   713→    var multiColRows = splitRows.filter(function (r) { return r.length >= 2; }).length;
   714→   714→    if (multiColRows < splitRows.length * 0.6) return { rows: [], columns: [] };
   715→   715→    var headers = splitRows[0].map(function (h, idx) { return h || 'col_' + (idx + 1); });
   716→   716→    var rows = splitRows.slice(1).filter(function (r) { return r.length; }).map(function (r) {
   717→   717→      var o = {};
   718→   718→      headers.forEach(function (h, idx) {
   719→   719→        var v = r[idx] != null ? r[idx] : '';
   720→   720→        if (v !== '' && !isNaN(v) && /^-?\d+(\.\d+)?$/.test(v.replace(/[,$]/g, ''))) {
   721→   721→          o[h] = Number(v.replace(/[,$]/g, ''));
   722→   722→        } else {
   723→   723→          o[h] = v;
   724→   724→        }
   725→   725→      });
   726→   726→      return o;
   727→   727→    });
   728→   728→    return { rows: rows, columns: headers };
   729→   729→  }

   731→   731→  // Dispatcher: route a File to the right parser by extension + MIME.
   732→   732→  function parseFile(file) {
   733→   733→    var name = (file.name || '').toLowerCase();

   735→   735→      return Promise.reject(new Error('For demo performance, please use files under 5 MB.'));

   779→   779→  // ─── Groq auto-analysis ────────────────────────────────────────────────────
   780→   780→  // Sends a sample of the document to Groq and parses the JSON analysis. Result
   781→   781→  // shape: { documentType, summary, keyColumns, suggestedActions, detectedFormulas, dataQualityNotes }
   782→   782→  function analyzeDocument(doc) {
   783→   783→    var sample = doc.rows && doc.rows.length
   784→   784→      ? JSON.stringify(doc.rows.slice(0, 5), null, 2)
   785→   785→      : JSON.stringify((doc.rawText || '').slice(0, 800));

  1501→  1501→  // Returns an HTML string. Numeric columns detected by scanning the first
  1502→  1502→  // non-empty cell of each column — right-aligned + number-formatted.
  1503→  1503→  function detectNumericColumns(rows, columns) {
  1504→  1504→    var isNum = {};
  1505→  1505→    columns.forEach(function (c) {
  1506→  1506→      var found = false;
  1507→  1507→      for (var i = 0; i < rows.length && i < 20; i++) {
  1508→  1508→        var v = rows[i][c];
  1509→  1509→        if (v !== '' && v != null) {
  1510→  1510→          isNum[c] = typeof v === 'number' || (!isNaN(Number(v)) && /^-?\d+(\.\d+)?$/.test(String(v).replace(/[,$]/g, '')));
  1511→  1511→          found = true;
  1512→  1512→          break;
  1513→  1513→        }
  1514→  1514→      }
  1515→  1515→      if (!found) isNum[c] = false;
  1516→  1516→    });
  1517→  1517→    return isNum;
  1518→  1518→  }

  1547→  1547→  // ─── ECharts chart renderer ──────────────────────────────────────────────────
  1548→  1548→  // Auto-picks chart type from data shape:
  1549→  1549→  //   • 1 row → no chart
  1550→  1550→  //   • 2 cols, col1 categorical + col2 numeric → bar
  1551→  1551→  //   • 2 cols, col1 looks like a date → line (sorted by date)
  1552→  1552→  //   • 3+ cols, col1 categorical + rest numeric → grouped bar
  1553→  1553→  //   • otherwise → "chart not available" message
  1554→  1554→  function looksLikeDateColumn(rows, col) {
  1555→  1555→    if (!rows.length) return false;
  1556→  1556→    var parsed = 0, total = 0;
  1557→  1557→    for (var i = 0; i < Math.min(rows.length, 12); i++) {
  1558→  1558→      var v = rows[i][col];
  1559→  1559→      if (v == null || v === '') continue;
  1560→  1560→      total++;
  1561→  1561→      var d = new Date(v);
  1562→  1562→      if (!isNaN(d.getTime()) && d.getFullYear() > 1990) parsed++;
  1563→  1563→    }
  1564→  1564→    return total >= 3 && parsed / total >= 0.7;
  1565→  1565→  }
  1566→  1566→  function isNumericVal(v) {
  1567→  1567→    if (typeof v === 'number') return true;
  1568→  1568→    return v != null && v !== '' && !isNaN(Number(String(v).replace(/[,$]/g, ''))) && /^-?\d+(\.\d+)?$/.test(String(v).replace(/[,$]/g, ''));
  1569→  1569→  }

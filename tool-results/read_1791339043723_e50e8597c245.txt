   243→  // ─── JSON extraction from Groq output ──────────────────────────────────────
   244→  // Groq often wraps JSON in ```json ... ``` fences despite the prompt asking
   245→  // for raw JSON. Strip fences + any leading/trailing prose, then JSON.parse.
   246→  function extractJson(content) {
   247→    if (!content) throw new Error('AI returned an empty response. Try rephrasing your instruction.');
   248→    var s = String(content).trim();
   249→    // Strip markdown code fences if present.
   250→    var fence = s.match(/```(?:json)?\s*([\s\S]*?)```/i);
   251→    if (fence) s = fence[1].trim();
   252→    // If there's still prose before/after the JSON object, try to isolate the
   253→    // first { ... } block.
   254→    if (s.charAt(0) !== '{' && s.charAt(0) !== '[') {
   255→      var start = s.indexOf('{');
   256→      if (start < 0) start = s.indexOf('[');
   257→      if (start >= 0) s = s.slice(start);
   258→    }
   259→    var endBrace = s.lastIndexOf('}');
   260→    var endBracket = s.lastIndexOf(']');
   261→    var end = Math.max(endBrace, endBracket);
   262→    if (end > 0 && end < s.length - 1) s = s.slice(0, end + 1);
   263→    try {
   264→      return JSON.parse(s);
   265→    } catch (e) {
   266→      throw new Error('The AI returned an unexpected response. Try rephrasing your instruction.');
   267→    }
   268→  }

   440→  // CSV — native parse, no library. Handles quoted fields with embedded commas,
   441→  // newlines, and doubled-quote escapes (RFC 4180).
   442→  function parseCSV(file) {
   443→    return new Promise(function (resolve, reject) {
   444→      var reader = new FileReader();
   445→      reader.onerror = function () { reject(new Error('Couldn\u2019t read this file.')); };
   446→      reader.onload = function () {
   447→        try {
   448→          var text = reader.result;
   449→          var rows = csvToRows(text);
   450→          if (!rows.length) {
   451→            reject(new Error('Couldn\u2019t parse this CSV — the file appears to be empty.'));
   452→            return;
   453→          }
   454→          resolve(makeDoc({
   455→            type: 'csv', name: file.name, fileSize: file.size,
   456→            rows: rows.rows, columns: rows.columns, rawText: ''
   457→          }));
   458→        } catch (e) {
   459→          reject(new Error('Couldn\u2019t parse this CSV. Make sure it\u2019s a valid .csv and try again.'));
   460→        }
   461→      };
   462→      reader.readAsText(file);
   463→    });

   465→  // Minimal RFC-4180 CSV parser → { rows: [{...}], columns: [...] }
   466→  function csvToRows(text) {
   467→    var records = [];
   468→    var field = '';
   469→    var row = [];
   470→    var i = 0;
   471→    var inQuotes = false;
   472→    while (i < text.length) {
   473→      var c = text[i];
   474→      if (inQuotes) {
   475→        if (c === '"') {
   476→          if (text[i + 1] === '"') { field += '"'; i += 2; continue; }
   477→          inQuotes = false; i++; continue;
   478→        }
   479→        field += c; i++; continue;
   480→      }
   481→      if (c === '"') { inQuotes = true; i++; continue; }
   482→      if (c === ',') { row.push(field); field = ''; i++; continue; }
   483→      if (c === '\r') { i++; continue; }
   484→      if (c === '\n') { row.push(field); records.push(row); row = []; field = ''; i++; continue; }
   485→      field += c; i++;
   486→    }
   487→    if (field.length || row.length) { row.push(field); records.push(row); }
   488→    if (!records.length) return { rows: [], columns: [] };
   489→    var headers = records[0].map(function (h, idx) { return h || ('col_' + (idx + 1)); });
   490→    var objRows = records.slice(1).filter(function (r) {
   491→      return r.some(function (v) { return v !== '' && v != null; });
   492→    }).map(function (r) {
   493→      var o = {};
   494→      headers.forEach(function (h, idx) {
   495→        var v = r[idx];
   496→        // Auto-coerce numeric strings — keeps summarize/sort working.
   497→        if (v != null && v !== '' && !isNaN(v) && /^-?\d+(\.\d+)?$/.test(v.trim())) o[h] = Number(v);
   498→        else o[h] = v != null ? v : '';
   499→      });
   500→      return o;
   501→    });
   502→    return { rows: objRows, columns: headers };
   503→  }

   546→  // <table> for tabular data. If no table found, treat as text-only doc.

   549→      return new Promise(function (resolve, reject) {
   550→        var reader = new FileReader();
   551→        reader.onerror = function () { reject(new Error('Couldn\u2019t read this Word file.')); };
   552→        reader.onload = function () {
   553→          var ab = reader.result;
   554→          // extractRawText + convertToHtml in parallel.

   556→            mammoth.extractRawText({ arrayBuffer: ab }).promise ||
   557→              mammoth.extractRawText({ arrayBuffer: ab }),
   558→            mammoth.convertToHtml({ arrayBuffer: ab }).promise ||
   559→              mammoth.convertToHtml({ arrayBuffer: ab })

   576→              }));

   579→            reject(new Error('Couldn\u2019t parse this Word file. Make sure it\u2019s a valid .docx (legacy .doc is not supported) and try again.'));
   580→          });
   581→        };
   582→        reader.readAsArrayBuffer(file);
   583→      });
   584→    });

   586→  // Parse the first <table> in an HTML string into { rows, columns }.
   587→  function parseFirstHtmlTable(html) {
   588→    if (!html) return { rows: [], columns: [] };
   589→    var tmp = document.createElement('div');
   590→    tmp.innerHTML = html;
   591→    var table = tmp.querySelector('table');
   592→    if (!table) return { rows: [], columns: [] };
   593→    var trs = table.querySelectorAll('tr');
   594→    if (!trs.length) return { rows: [], columns: [] };
   595→    var headers = [];
   596→    var firstCells = trs[0].querySelectorAll('th,td');
   597→    firstCells.forEach(function (c, idx) { headers.push((c.textContent || '').trim() || 'col_' + (idx + 1)); });
   598→    // If first row has <th>, treat it as header; otherwise the first row is data.
   599→    var startIdx = trs[0].querySelector('th') ? 1 : 0;
   600→    var rows = [];
   601→    for (var i = startIdx; i < trs.length; i++) {
   602→      var cells = trs[i].querySelectorAll('td,th');
   603→      if (!cells.length) continue;
   604→      var o = {};
   605→      headers.forEach(function (h, idx) {
   606→        var v = (cells[idx] ? cells[idx].textContent : '').trim();
   607→        if (v !== '' && !isNaN(v) && /^-?\d+(\.\d+)?$/.test(v)) o[h] = Number(v);
   608→        else o[h] = v;
   609→      });
   610→      rows.push(o);
   611→    }
   612→    return { rows: rows, columns: headers };
   613→  }

   615→  // PDF (.pdf) via pdf.js. Extracts text per page. Attempts a simple table
   616→  // detection: if multiple text items share the same Y coordinate across pages,
   617→  // group them into rows. If detection fails, fall back to text-only.
   618→  function parsePDF(file) {
   619→    return loadScript('pdfjs').then(function () {
   620→      return new Promise(function (resolve, reject) {
   621→        var reader = new FileReader();
   622→        reader.onerror = function () { reject(new Error('Couldn\u2019t read this PDF.')); };
   623→        reader.onload = function () {
   624→          try {
   625→            // Configure the worker — pdf.js won't run without it.
   626→            if (window.pdfjsLib) {
   627→              pdfjsLib.GlobalWorkerOptions.workerSrc = LIB_URLS.pdfjsWorker;
   628→            }
   629→            var loadingTask = pdfjsLib.getDocument({ data: reader.result });
   630→            loadingTask.promise.then(function (pdf) {
   631→              var numPages = pdf.numPages;
   632→              var allItems = [];   // { page, str, x, y, w }
   633→              var rawText  = [];
   634→              // Sequentially fetch each page's text content. We do this serially
   635→              // (rather than Promise.all) because pdf.js's getTextContent can
   636→              // interfere with concurrent page operations on some builds, and
   637→              // serial is plenty fast for the demo's <5MB PDFs.
   638→              function readPagesSequentially(pageNum) {
   639→                if (pageNum > numPages) return Promise.resolve();
   640→                return pdf.getPage(pageNum).then(function (page) {
   641→                  return page.getTextContent().then(function (tc) {
   642→                    (tc.items || []).forEach(function (it) {
   643→                      allItems.push({
   644→                        page: pageNum,
   645→                        str: it.str,
   646→                        x: it.transform ? it.transform[4] : 0,
   647→                        y: it.transform ? it.transform[5] : 0,
   648→                        w: it.width || 0
   649→                      });
   650→                      if (it.str) rawText.push(it.str);
   651→                    });
   652→                    return readPagesSequentially(pageNum + 1);
   653→                  });
   654→                });
   655→              }
   656→              return readPagesSequentially(1).then(function () {
   657→                var text = rawText.join(' ').replace(/\s+/g, ' ').trim();
   658→                var parsed = detectPDFTable(allItems);
   659→                var warnings = [];
   660→                if (parsed.rows.length) {
   661→                  warnings.push('Detected a tabular layout from text positioning — rows may be imperfect. Verify against the source PDF.');
   662→                } else if (text) {
   663→                  warnings.push('No clear table layout detected — treating as a text document. Pipeline steps that need rows will return empty results, but the AI can still analyze the text.');
   664→                }
   665→                resolve(makeDoc({
   666→                  type: 'pdf', name: file.name, fileSize: file.size,
   667→                  rows: parsed.rows, columns: parsed.columns,
   668→                  rawText: text || '(no extractable text)', warnings: warnings
   669→                }));
   670→              });
   671→            }).catch(function () {
   672→              reject(new Error('Couldn\u2019t open this PDF. The file may be corrupted, password-protected, or a scanned image (text-extractable PDFs only).'));
   673→            });
   674→          } catch (e) {
   675→            reject(new Error('Couldn\u2019t parse this PDF. Make sure it\u2019s a valid .pdf and try again.'));
   676→          }
   677→        };
   678→        reader.readAsArrayBuffer(file);
   679→      });
   680→    });
   681→  }

   683→  // Heuristic PDF table detector: group items by Y coordinate (rounded to
   684→  // nearest 2px to absorb sub-pixel jitter), sort each row's items by X, and
   685→  // join into cells. If at least 3 rows × 2 cols, treat as a table.
   686→  function detectPDFTable(items) {
   687→    if (!items || items.length < 6) return { rows: [], columns: [] };
   688→    // Group by rounded Y.
   689→    var byY = {};
   690→    items.forEach(function (it) {
   691→      if (!it.str) return;
   692→      var y = Math.round(it.y / 2) * 2;
   693→      if (!byY[y]) byY[y] = [];
   694→      byY[y].push(it);
   695→    });
   696→    var yKeys = Object.keys(byY).map(Number).sort(function (a, b) { return b - a; }); // top of page = larger Y
   697→    if (yKeys.length < 3) return { rows: [], columns: [] };
   698→    // Build row arrays (items sorted by X within each row).
   699→    var rowArrays = yKeys.map(function (y) {
   700→      return byY[y].sort(function (a, b) { return a.x - b.x; })
   701→        .map(function (it) { return it.str; })
   702→        .join(' ').trim();
   703→    }).filter(function (s) { return s.length; });
   704→    if (rowArrays.length < 3) return { rows: [], columns: [] };
   705→    // Split each row on 2+ space gaps into columns. If the first row splits into
   706→    // ≥2 columns, use that count as the column count.
   707→    var splitRows = rowArrays.map(function (r) {
   708→      return r.split(/\s{2,}/).map(function (s) { return s.trim(); }).filter(function (s) { return s.length; });
   709→    });
   710→    var colCount = splitRows[0].length;
   711→    if (colCount < 2) return { rows: [], columns: [] };
   712→    // Require that most rows have ≥2 columns; otherwise it's just paragraph text.
   713→    var multiColRows = splitRows.filter(function (r) { return r.length >= 2; }).length;
   714→    if (multiColRows < splitRows.length * 0.6) return { rows: [], columns: [] };
   715→    var headers = splitRows[0].map(function (h, idx) { return h || 'col_' + (idx + 1); });
   716→    var rows = splitRows.slice(1).filter(function (r) { return r.length; }).map(function (r) {
   717→      var o = {};
   718→      headers.forEach(function (h, idx) {
   719→        var v = r[idx] != null ? r[idx] : '';
   720→        if (v !== '' && !isNaN(v) && /^-?\d+(\.\d+)?$/.test(v.replace(/[,$]/g, ''))) {
   721→          o[h] = Number(v.replace(/[,$]/g, ''));
   722→        } else {
   723→          o[h] = v;
   724→        }
   725→      });
   726→      return o;
   727→    });
   728→    return { rows: rows, columns: headers };
   729→  }

   813→  // ─── Groq pipeline-spec generator (NL → JSON) ──────────────────────────────
   814→  function generatePipelineSpec(doc, instruction) {
   815→    var sampleRow = doc.rows && doc.rows.length ? JSON.stringify(doc.rows[0]) : '(no rows)';
   816→    var prompt =
   817→      'You are an AI pipeline constructor. Convert the user\'s natural language instruction into a JSON pipeline spec.\n\n' +
   818→      'User\'s instruction: "' + instruction + '"\n\n' +
   819→      'Document context:\n' +
   820→      '- Type: ' + doc.type + '\n' +
   821→      '- Name: ' + doc.name + '\n' +
   822→      '- Columns: ' + (doc.columns.join(', ') || '(none)') + '\n' +
   823→      '- Sample row: ' + sampleRow + '\n' +
   824→      '- Row count: ' + doc.rowCount + '\n\n' +
   825→      'Respond with a JSON object ONLY (no markdown, no explanation):\n' +
   826→      '{\n' +
   827→      '  "name": "short pipeline name",\n' +
   828→      '  "description": "one-sentence description of what the pipeline does",\n' +
   829→      '  "steps": [\n' +
   830→      '    { "type": "filter"|"summarize"|"sort"|"limit"|"select"|"transform", "description": "what this step does",\n' +
   831→      '      // filter:    { "column": "...", "operator": "=="|"!="|">"|"<"|">="|"<="|"contains", "value": ... }\n' +
   832→      '      // summarize: { "groupBy": ["col1", ...], "aggregation": "sum"|"count"|"avg"|"min"|"max", "valueColumn": "..." }\n' +
   833→      '      // sort:      { "column": "...", "order": "asc"|"desc" }\n' +
   834→      '      // limit:     { "count": 5 }\n' +
   835→      '      // select:    { "columns": ["col1", "col2"] }\n' +
   836→      '      // transform: { "newColumn": "...", "expression": "qty * price" } }\n' +
   837→      '    }\n' +
   838→      '  ],\n' +
   839→      '  "output": { "format": "html_table"|"csv"|"excel"|"pdf"|"html_text", "emailSubject": "subject line", "emailBodyIntro": "one-sentence intro" }\n' +
   840→      '}\n\n' +
   841→      'CRITICAL DISPATCH RULE:\n' +
   842→      '  - If the instruction is a TABULAR operation (filter, summarize, aggregate, sort, count, group, top N, etc.) -> return spec with "steps[]"\n' +
   843→      '  - If the instruction is a TEXT TASK (summarize the document, main points, key takeaways, what does this report say, ask a question, explain, describe) -> return spec with "aiTask{}" instead of "steps[]"\n' +
   844→      '  - DO NOT include both. Pick one based on the instruction wording.\n\n' +
   845→      'aiTask shape (only when the instruction is a text task):\n' +
   846→      '  "aiTask": {\n' +
   847→      '    "type": "summary" | "bullet_points" | "qa",\n' +
   848→      '    "question": "...",          // ONLY for type="qa" — restate the user\'s question\n' +
   849→      '    "length": "short" | "medium" | "long",   // optional, only for type="summary"\n' +
   850→      '    "count": 5                  // optional, only for type="bullet_points" — how many points to extract\n' +
   851→      '  }\n' +
   852→      '  output.format for aiTask = "html_text"\n' +
   853→      '  Leave "steps" OUT of the JSON entirely when aiTask is present.';
   854→    return callGroq(
   855→      [{ role: 'system', content: 'You are a pipeline spec generator that outputs only raw JSON.' },
   856→       { role: 'user', content: prompt }],
   857→      { temperature: 0.2, max_tokens: 1000 }
   858→    ).then(function (content) {
   859→      var spec = extractJson(content);
   860→      // Accept EITHER steps[] (tabular) OR aiTask{} (text task).
   861→      // Groq decides which path based on the instruction wording.
   862→      var hasAiTask = spec.aiTask && typeof spec.aiTask === 'object' && typeof spec.aiTask.type === 'string';
   863→      var hasSteps = spec.steps && Array.isArray(spec.steps) && spec.steps.length;
   864→      if (!hasAiTask && !hasSteps) {
   865→        throw new Error('The AI didn\u2019t produce a pipeline or an AI task. Try rephrasing your instruction.');
   866→      }
   867→      if (hasSteps && !hasAiTask) {

   871→          throw new Error('The AI returned an unexpected response. Try rephrasing your instruction.');
   872→        }
   873→      }
   874→      if (!spec.output || typeof spec.output !== 'object') {
   875→        spec.output = { format: hasAiTask ? 'html_text' : 'html_table' };
   876→      }

   881→  // ─── Safe expression evaluator (recursive descent) ─────────────────────────
   882→  // ─────────────────────────────────────────────────────────────────────────────
   883→  // Used by the "transform" step to evaluate expressions like "qty * price" or
   884→  // "first_name + ' ' + last_name" WITHOUT eval() or new Function() — those are
   885→  // XSS vectors and CSP-violators. This parser supports:
   886→  //   • numeric literals (12, 3.14)
   887→  //   • single-quoted string literals ('hello')
   888→  //   • column identifiers (qty, price, first_name)
   889→  //   • operators: + - * / %  (binary, left-associative; + on a string operand
   890→  //     coerces both sides to string and concatenates)
   891→  //   • parentheses for grouping
   892→  // Anything else throws — fail closed.
   893→  //
   894→  // Grammar (EBNF):
   895→  //   expression := term { ('+' | '-') term }
   896→  //   term       := factor { ('*' | '/' | '%') factor }
   897→  //   factor     := number | string | identifier | '(' expression ')'
   898→  // ─────────────────────────────────────────────────────────────────────────────
   899→
   900→  function tokenizeExpr(expr) {
   901→    var tokens = [];
   902→    var i = 0, len = expr.length;
   903→    while (i < len) {
   904→      var c = expr[i];
   905→      if (c === ' ' || c === '\t' || c === '\n' || c === '\r') { i++; continue; }
   906→      // Number literal — digits and at most one decimal point.
   907→      if ((c >= '0' && c <= '9') || c === '.') {
   908→        var num = '';
   909→        var dotSeen = false;
   910→        while (i < len && ((expr[i] >= '0' && expr[i] <= '9') || expr[i] === '.')) {
   911→          if (expr[i] === '.') {
   912→            if (dotSeen) break;
   913→            dotSeen = true;
   914→          }
   915→          num += expr[i]; i++;
   916→        }
   917→        var n = parseFloat(num);
   918→        if (isNaN(n)) throw new Error('Invalid number in expression: ' + num);
   919→        tokens.push({ type: 'num', value: n });
   920→        continue;
   921→      }
   922→      // Single-quoted string literal — doubled '' inside means an escaped quote.
   923→      if (c === "'") {
   924→        var str = '';
   925→        i++; // skip opening quote
   926→        while (i < len && expr[i] !== "'") { str += expr[i]; i++; }
   927→        if (i >= len) throw new Error('Unterminated string literal in expression');
   928→        i++; // skip closing quote
   929→        tokens.push({ type: 'str', value: str });
   930→        continue;
   931→      }
   932→      // Identifier — letters, digits, underscore (must start with letter/underscore).
   933→      if ((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || c === '_') {
   934→        var id = '';
   935→        while (i < len && ((expr[i] >= 'a' && expr[i] <= 'z') ||
   936→                           (expr[i] >= 'A' && expr[i] <= 'Z') ||
   937→                           (expr[i] >= '0' && expr[i] <= '9') || expr[i] === '_')) {
   938→          id += expr[i]; i++;
   939→        }
   940→        tokens.push({ type: 'id', value: id });
   941→        continue;
   942→      }
   943→      // Operators + parens.
   944→      if ('+-*/%()'.indexOf(c) >= 0) { tokens.push({ type: 'op', value: c }); i++; continue; }
   945→      throw new Error('Invalid character in expression: ' + c);
   946→    }
   947→    return tokens;
   948→  }
   949→
   950→  // Recursive-descent parser → AST nodes:
   951→  //   { type: 'literal', value: <number|string> }
   952→  //   { type: 'column',  name: <string> }
   953→  //   { type: 'binop',   op: '+'|'-'|'*'|'/'|'%', left: <node>, right: <node> }
   954→  function parseExpr(tokens) {
   955→    var pos = 0;
   956→    function peek() { return tokens[pos]; }
   957→    function next() { return tokens[pos++]; }
   958→    function parseExpression() {
   959→      var node = parseTerm();
   960→      while (peek() && peek().type === 'op' && (peek().value === '+' || peek().value === '-')) {
   961→        var op = next().value;
   962→        var right = parseTerm();
   963→        node = { type: 'binop', op: op, left: node, right: right };
   964→      }
   965→      return node;
   966→    }
   967→    function parseTerm() {
   968→      var node = parseFactor();
   969→      while (peek() && peek().type === 'op' &&
   970→             (peek().value === '*' || peek().value === '/' || peek().value === '%')) {
   971→        var op = next().value;
   972→        var right = parseFactor();
   973→        node = { type: 'binop', op: op, left: node, right: right };
   974→      }
   975→      return node;
   976→    }
   977→    function parseFactor() {
   978→      var t = peek();
   979→      if (!t) throw new Error('Unexpected end of expression');
   980→      if (t.type === 'num' || t.type === 'str') { next(); return { type: 'literal', value: t.value }; }
   981→      if (t.type === 'id') { next(); return { type: 'column', name: t.value }; }
   982→      if (t.type === 'op' && t.value === '(') {
   983→        next();
   984→        var node = parseExpression();
   985→        if (!peek() || peek().value !== ')') throw new Error('Missing closing parenthesis');
   986→        next(); // consume ')'
   987→        return node;
   988→      }
   989→      throw new Error('Unexpected token in expression: ' + t.value);
   990→    }
   991→    var ast = parseExpression();
   992→    if (pos < tokens.length) throw new Error('Unexpected trailing tokens in expression');
   993→    return ast;
   994→  }
   995→
   996→  // Evaluate an AST node against a row object. The '+' operator is polymorphic:
   997→  // if either operand is a string, both sides stringify and concatenate; else
   998→  // numeric addition. All other operators are numeric only.
   999→  function evalAst(node, row) {
  1000→    if (node.type === 'literal') return node.value;
  1001→    if (node.type === 'column') {
  1002→      if (!(node.name in row)) throw new Error('Unknown column in expression: ' + node.name);
  1003→      return row[node.name];
  1004→    }
  1005→    if (node.type === 'binop') {
  1006→      var l = evalAst(node.left, row);
  1007→      var r = evalAst(node.right, row);
  1008→      if (node.op === '+' && (typeof l === 'string' || typeof r === 'string')) {
  1009→        return String(l) + String(r);
  1010→      }
  1011→      var ln = Number(l), rn = Number(r);
  1012→      if (isNaN(ln) || isNaN(rn)) throw new Error('Non-numeric operand in arithmetic expression');
  1013→      switch (node.op) {
  1014→        case '+': return ln + rn;
  1015→        case '-': return ln - rn;
  1016→        case '*': return ln * rn;
  1017→        case '/': return rn === 0 ? 0 : ln / rn;
  1018→        case '%': return rn === 0 ? 0 : ln % rn;
  1019→      }
  1020→      throw new Error('Unknown operator: ' + node.op);
  1021→    }
  1022→    throw new Error('Cannot evaluate AST node: ' + node.type);
  1023→  }
  1024→
  1025→  // Public façade — tokenize + parse + evaluate. AST is cached per expression
  1026→  // (via the closure on the parsed result) so per-row evaluation skips parsing.
  1027→  function makeExprEvaluator(expression) {
  1028→    var tokens = tokenizeExpr(expression);
  1029→    var ast = parseExpr(tokens);
  1030→    return function (row) { return evalAst(ast, row); };
  1031→  }

  1033→  // ─── Case-insensitive column resolver ──────────────────────────────────────
  1034→  // Groq frequently returns column names in a DIFFERENT CASE than the actual
  1035→  // data — e.g. valueColumn: "QTY" when the data has lowercase "qty", or
  1036→  // "Revenue" when the data has "revenue". Without a resolver, r["QTY"]
  1037→  // returns undefined → Number(undefined) is NaN → NaN propagates through
  1038→  // sum/avg/min/max → the entire result column shows as NaN.
  1039→  //
  1040→  // The resolver: exact match first, case-insensitive match second, falls
  1041→  // back to the original name (so undefined values surface the bug visibly
  1042→  // in the result — better than silently failing).
  1043→  var _colLookupCache = {};
  1044→  function resolveColumn(name, docColumns) {
  1045→    if (!name) return name;
  1046→    if (docColumns.indexOf(name) >= 0) return name;
  1047→    var cacheKey = name.toLowerCase() + '\u0001' + docColumns.join('\u0001');
  1048→    if (_colLookupCache[cacheKey]) return _colLookupCache[cacheKey];
  1049→    var lower = name.toLowerCase();
  1050→    for (var i = 0; i < docColumns.length; i++) {
  1051→      if (docColumns[i].toLowerCase() === lower) {
  1052→        _colLookupCache[cacheKey] = docColumns[i];
  1053→        return docColumns[i];
  1054→      }
  1055→    }
  1056→    _colLookupCache[cacheKey] = name;
  1057→    return name;
  1058→  }

  1060→  // ─── AI text-task executor (summary / bullet_points / Q&A) ────────────────
  1061→  // Used when the spec has `aiTask` instead of `steps[]`. Calls Groq with a
  1062→  // task-specific prompt, using the document content as context. Returns a
  1063→  // string (the AI-generated summary / bulleted list / answer).
  1064→  //
  1065→  // Document content passed to Groq:
  1066→  //   - For row-based docs (Excel/CSV): first 30 rows as JSON
  1067→  //   - For text docs (PDF/Word): first 4000 chars of rawText

  1081→      docContent = '(empty document — no extractable content)';

  1083→
  1084→    var taskType = aiTask.type || 'summary';
  1085→    var systemPrompt, userPrompt;
  1086→

  1102→        'Output: one bullet per line, each starting with \u2022 (a bullet character). ' +
  1103→        'No markdown headers, no intro paragraph, just the bulleted list.';
  1104→      userPrompt = 'Document: ' + doc.name + ' (' + doc.type + ')\n\n' +
  1105→        docContent + '\n\n' +
  1106→        'List ' + count + ' main points:';
  1107→    } else {
  1108→      // summary (default)
  1109→      var lengthHint = aiTask.length === 'short' ? '1 paragraph (3-4 sentences)' :
  1110→                       aiTask.length === 'long' ? '3-4 paragraphs' :
  1111→                       '2 paragraphs (4-6 sentences total)';
  1112→      systemPrompt = 'You are an analyst writing a plain-English summary of a document. ' +
  1113→        'Capture the document\'s purpose, key data points, notable patterns, and any anomalies. ' +
  1114→        'Write ' + lengthHint + '. ' +
  1115→        'Plain text, no markdown, no headers, just paragraphs.';

  1128→  }

  1130→  // ─── Pipeline executor ─────────────────────────────────────────────────────
  1131→  // Runs each step in order over a shallow-cloned row array. Each step mutates
  1132→  // the array (filter shortens, summarize reshapes, sort reorders, etc.). The
  1133→  // final shape is returned to the renderer.
  1134→  function executePipeline(doc, spec) {
  1135→    var rows = (doc.rows || []).slice();
  1136→    var docCols = doc.columns || (rows.length ? Object.keys(rows[0]) : []);
  1137→    _colLookupCache = {};  // reset per-execution
  1138→    (spec.steps || []).forEach(function (step) {
  1139→      if (step.type === 'filter') {
  1140→        var fCol = resolveColumn(step.column, docCols);
  1141→        rows = rows.filter(function (r) {
  1142→          var v = r[fCol];
  1143→          switch (step.operator) {
  1144→            case '==':  return String(v) == String(step.value);
  1145→            case '!=':  return String(v) != String(step.value);
  1146→            case '>':   return Number(v) >  Number(step.value);
  1147→            case '<':   return Number(v) <  Number(step.value);
  1148→            case '>=':  return Number(v) >= Number(step.value);
  1149→            case '<=':  return Number(v) <= Number(step.value);
  1150→            case 'contains':
  1151→              return String(v).toLowerCase().indexOf(String(step.value).toLowerCase()) >= 0;
  1152→            default: return true;
  1153→          }
  1154→        });
  1155→      } else if (step.type === 'summarize') {
  1156→        var groupByRaw = step.groupBy && step.groupBy.length ? step.groupBy : [];
  1157→        var groupBy = groupByRaw.map(function (c) { return resolveColumn(c, docCols); });
  1158→        var valCol = step.valueColumn ? resolveColumn(step.valueColumn, docCols) : null;
  1159→        var groups = {};
  1160→        var order = [];
  1161→        rows.forEach(function (r) {
  1162→          var key = groupBy.map(function (c) { return r[c]; }).join(' | ');
  1163→          if (!groups[key]) { groups[key] = { count: 0, sum: 0, min: Infinity, max: -Infinity }; order.push(key); }
  1164→          var g = groups[key];
  1165→          var val = valCol ? Number(r[valCol]) : 0;
  1166→          if (isNaN(val)) val = 0;  // guard against NaN propagation
  1167→          g.count++;
  1168→          if (valCol) {
  1169→            g.sum += val;
  1170→            g.min = Math.min(g.min, val);
  1171→            g.max = Math.max(g.max, val);
  1172→          }
  1173→        });
  1174→        rows = order.map(function (key) {
  1175→          var g = groups[key];
  1176→          var parts = key.split(' | ');
  1177→          var row = {};
  1178→          groupBy.forEach(function (c, i) { row[c] = parts[i]; });
  1179→          if (valCol) {
  1180→            var agg = step.aggregation || 'sum';
  1181→            row[agg + '_of_' + valCol] =
  1182→              agg === 'sum'   ? g.sum :
  1183→              agg === 'count' ? g.count :
  1184→              agg === 'avg'   ? (g.count ? g.sum / g.count : 0) :
  1185→              agg === 'min'   ? (g.min === Infinity ? 0 : g.min) :
  1186→              agg === 'max'   ? (g.max === -Infinity ? 0 : g.max) : null;
  1187→          } else {
  1188→            row.count = g.count;
  1189→          }
  1190→          return row;
  1191→        });
  1192→      } else if (step.type === 'sort') {
  1193→        var sCol = resolveColumn(step.column, docCols);
  1194→        var order2 = step.order === 'desc' ? -1 : 1;
  1195→        rows.sort(function (a, b) {
  1196→          var av = a[sCol], bv = b[sCol];
  1197→          if (typeof av === 'number' && typeof bv === 'number') return (av - bv) * order2;
  1198→          return order2 * String(av).localeCompare(String(bv));
  1199→        });
  1200→      } else if (step.type === 'limit') {
  1201→        rows = rows.slice(0, Math.max(0, step.count | 0));
  1202→      } else if (step.type === 'select') {
  1203→        var selColsRaw = step.columns || [];
  1204→        var selCols = selColsRaw.map(function (c) { return resolveColumn(c, docCols); });
  1205→        rows = rows.map(function (r) {
  1206→          var o = {};
  1207→          selCols.forEach(function (c) { o[c] = r[c]; });
  1208→          return o;
  1209→        });
  1210→      } else if (step.type === 'transform') {
  1211→        // Build the evaluator ONCE, then map — parsing per row would be wasteful.
  1212→        var evalFn;
  1213→        try { evalFn = makeExprEvaluator(step.expression); }
  1214→        catch (e) { throw new Error('Transform step has an invalid expression: ' + e.message); }
  1215→        var newCol = step.newColumn || 'computed';
  1216→        rows = rows.map(function (r) {
  1217→          var o = Object.assign({}, r);
  1218→          try { o[newCol] = evalFn(r); } catch (e) { o[newCol] = ''; }
  1219→          return o;
  1220→        });
  1221→      }
  1222→      // Unknown step types are silently skipped — the spec validator already
  1223→      // filtered out garbage, so this is a defensive no-op.
  1224→    });
  1225→    return { rows: rows, columns: rows.length ? Object.keys(rows[0]) : [] };
  1226→  }

  1500→  // ─── HTML table renderer ────────────────────────────────────────────────────
  1501→  // Returns an HTML string. Numeric columns detected by scanning the first
  1502→  // non-empty cell of each column — right-aligned + number-formatted.
  1503→  function detectNumericColumns(rows, columns) {
  1504→    var isNum = {};
  1505→    columns.forEach(function (c) {
  1506→      var found = false;
  1507→      for (var i = 0; i < rows.length && i < 20; i++) {
  1508→        var v = rows[i][c];
  1509→        if (v !== '' && v != null) {
  1510→          isNum[c] = typeof v === 'number' || (!isNaN(Number(v)) && /^-?\d+(\.\d+)?$/.test(String(v).replace(/[,$]/g, '')));
  1511→          found = true;
  1512→          break;
  1513→        }
  1514→      }
  1515→      if (!found) isNum[c] = false;
  1516→    });
  1517→    return isNum;
  1518→  }

  1520→  function renderTableHTML(rows, columns, visibleRows) {
  1521→    if (!rows.length) {
  1522→      return '<div class="pb-table-empty">No rows match this pipeline.</div>';
  1523→    }
  1524→    var isNum = detectNumericColumns(rows, columns);

  1527→    columns.forEach(function (c) { html += '<th class="' + (isNum[c] ? 'num' : '') + '">' + esc(c) + '</th>'; });
  1528→    html += '</tr></thead><tbody>';

  1530→      html += '<tr>';
  1531→      columns.forEach(function (c) {
  1532→        var v = r[c];
  1533→        var cell;
  1534→        if (isNum[c]) cell = fmtNumber(v);
  1535→        else cell = esc(v);

  1547→  // ─── ECharts chart renderer ──────────────────────────────────────────────────
  1548→  // Auto-picks chart type from data shape:
  1549→  //   • 1 row → no chart
  1550→  //   • 2 cols, col1 categorical + col2 numeric → bar
  1551→  //   • 2 cols, col1 looks like a date → line (sorted by date)
  1552→  //   • 3+ cols, col1 categorical + rest numeric → grouped bar
  1553→  //   • otherwise → "chart not available" message
  1554→  function looksLikeDateColumn(rows, col) {
  1555→    if (!rows.length) return false;
  1556→    var parsed = 0, total = 0;
  1557→    for (var i = 0; i < Math.min(rows.length, 12); i++) {
  1558→      var v = rows[i][col];
  1559→      if (v == null || v === '') continue;
  1560→      total++;
  1561→      var d = new Date(v);
  1562→      if (!isNaN(d.getTime()) && d.getFullYear() > 1990) parsed++;
  1563→    }
  1564→    return total >= 3 && parsed / total >= 0.7;
  1565→  }
  1566→  function isNumericVal(v) {
  1567→    if (typeof v === 'number') return true;
  1568→    return v != null && v !== '' && !isNaN(Number(String(v).replace(/[,$]/g, ''))) && /^-?\d+(\.\d+)?$/.test(String(v).replace(/[,$]/g, ''));
  1569→  }

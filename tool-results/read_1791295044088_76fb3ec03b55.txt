   422→// ─── Parse user-agent into short "Browser · OS" string ─────────────────────
   423→function parseDevice(ua) {
   424→  if (!ua || ua === 'unknown') return 'unknown';
   425→  var browser = 'Other';
   426→  if (ua.indexOf('Edg/') >= 0) browser = 'Edge';
   427→  else if (ua.indexOf('OPR/') >= 0) browser = 'Opera';
   428→  else if (ua.indexOf('Chrome/') >= 0) browser = 'Chrome';
   429→  else if (ua.indexOf('Firefox/') >= 0) browser = 'Firefox';
   430→  else if (ua.indexOf('Safari/') >= 0 && ua.indexOf('Chrome/') < 0) browser = 'Safari';
   431→  else if (ua.indexOf('curl') >= 0 || ua.indexOf('python') >= 0) browser = 'Bot';
   432→  var os = 'Other';
   433→  if (ua.indexOf('iPhone') >= 0) os = 'iPhone';
   434→  else if (ua.indexOf('iPad') >= 0) os = 'iPad';
   435→  else if (ua.indexOf('Android') >= 0) os = 'Android';
   436→  else if (ua.indexOf('Mac OS') >= 0 || ua.indexOf('Macintosh') >= 0) os = 'macOS';
   437→  else if (ua.indexOf('Windows') >= 0) os = 'Windows';
   438→  else if (ua.indexOf('Linux') >= 0) os = 'Linux';
   439→  return browser + ' · ' + os;
   440→}
   441→
   442→// Parse just the OS (strips browser — same device diff browser = same OS)
   443→function parseDeviceOS(ua) {
   444→  if (!ua || ua === 'unknown') return 'unknown';
   445→  if (ua.indexOf('iPhone') >= 0) return 'iPhone';
   446→  if (ua.indexOf('iPad') >= 0) return 'iPad';
   447→  if (ua.indexOf('Android') >= 0) return 'Android';
   448→  if (ua.indexOf('Mac OS') >= 0 || ua.indexOf('Macintosh') >= 0) return 'macOS';
   449→  if (ua.indexOf('Windows') >= 0) return 'Windows';
   450→  if (ua.indexOf('Linux') >= 0) return 'Linux';
   451→  return 'Other';
   452→}

   495→  function renderVisits(data) {
   496→    // Parse + enrich each visit with device (OS only — not Browser·OS, so same device with diff browser = 1)
   497→    var visits = data.visits.map(function(v) {
   498→      return {
   499→        ts: v.ts ? new Date(v.ts).getTime() : 0,
   500→        tsLabel: v.ts ? new Date(v.ts).toLocaleTimeString([], {month:'short',day:'numeric',hour:'2-digit',minute:'2-digit',second:'2-digit'}) : '—',
   501→        ip: v.ip || 'unknown',
   502→        loc: (v.city && v.city !== 'unknown' ? v.city + ', ' : '') + (v.country || 'unknown'),
   503→        page: v.page || '—',
   504→        device: parseDevice(v.ua),  // Browser · OS (for the table display)
   505→        os: parseDeviceOS(v.ua)    // OS only (for the distinct-device count)
   506→      };
   507→    });
   508→
   509→    // ─── Group: same device (IP + browser/OS) + same page within 5 min ─────
   510→    var SESSION_MS = 5 * 60 * 1000;
   511→    // Build groups keyed by ip||device||page → sorted visit arrays
   512→    var groups = {};
   513→    visits.forEach(function(v) {
   514→      var key = v.ip + '|' + v.device + '|' + v.page;
   515→      if (!groups[key]) groups[key] = [];
   516→      groups[key].push(v);
   517→    });
   518→    // Within each group, sort by timestamp, then split into sessions (5-min window)
   519→    var sessions = [];
   520→    Object.keys(groups).forEach(function(key) {
   521→      var arr = groups[key].sort(function(a, b) { return b.ts - a.ts; }); // newest first
   522→      var current = null;
   523→      arr.forEach(function(v) {
   524→        if (current && (current.ts - v.ts) <= SESSION_MS) {
   525→          // Same session — add to current group
   526→          current.reloads++;
   527→          current.lastTs = v.ts;
   528→          current.lastLabel = v.tsLabel;
   529→        } else {
   530→          // New session
   531→          if (current) sessions.push(current);
   532→          current = {
   533→            ts: v.ts, tsLabel: v.tsLabel,
   534→            ip: v.ip, loc: v.loc, page: v.page, device: v.device,
   535→            reloads: 1,
   536→            lastTs: v.ts, lastLabel: v.tsLabel
   537→          };
   538→        }
   539→      });
   540→      if (current) sessions.push(current);
   541→    });
   542→    // Sort sessions newest-first
   543→    sessions.sort(function(a, b) { return b.ts - a.ts; });
   544→
   545→    // Compute distinct IPs + devices from the RAW visit log (not deduped)
   546→    var ipMap = {};
   547→    var devMap = {};
   548→    visits.forEach(function(v) {
   549→      ipMap[v.ip] = (ipMap[v.ip] || 0) + 1;
   550→      devMap[v.os] = (devMap[v.os] || 0) + 1;
   551→    });
   552→
   553→    // Update summary cards — use the SAME recent visit log as the table
   554→    // (not the cumulative blob) so cards match the table.
   555→    // Show the cumulative (all-time) count as a separate footnote.
   556→    var cum = data.cumulative || {};
   557→    var recentIPs = Object.keys(ipMap).length;
   558→    var recentDevices = Object.keys(devMap).length;

   560→    document.getElementById('statTotal').textContent = recentTotal;
   561→    document.getElementById('statIPs').textContent = recentIPs;
   562→    document.getElementById('statDevices').textContent = recentDevices;
   563→    document.getElementById('statUpdated').textContent = '· ' + new Date().toLocaleTimeString() +
   564→      ' · all-time: ' + (cum.totalVisits || 0) + ' visits, ' + (cum.distinctIPs || 0) + ' IPs, ' + (cum.distinctDevices || 0) + ' devices';
   565→
   566→    // Device breakdown — from the same recent visit log as the table above

   568→    breakdownHTML += '<div style="font-size:9px;font-weight:700;text-transform:uppercase;letter-spacing:0.05em;color:var(--muted,#94a3b8);margin-bottom:8px;">Recent visitors (IP + OS) — ' + recentIPs + ' unique since last deploy</div>';
   569→    // Build IP+OS groups from the recent visit log (best available without edge function redeploy)
   570→    var allVisits = data.visits || [];

   586→      breakdownHTML += '<div style="display:flex;justify-content:space-between;gap:8px;padding:3px 0;border-bottom:1px solid rgba(0,0,0,0.03);line-height:1.4;font-size:11px;">';
